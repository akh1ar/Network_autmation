import paramiko
from netmiko import ConnectHandler

paramiko.Transport._preferred_kex = (
    "diffie-hellman-group14-sha1",
    *paramiko.Transport._preferred_kex
)
paramiko.Transport._preferred_keys = (
    "ssh-rsa",
    *paramiko.Transport._preferred_keys
)

cisco_SW1 = {
    "device_type": "cisco_ios",
    "ip": "192.168.1.1",
    "username": "cisco",
    "password": "cisco12345",
}

cisco_R1 = {
    "device_type": "cisco_ios",
    "ip": "192.168.1.2",
    "username": "admin",
    "password": "cisco12345",
}

#all_login = [cisco_SW1, cisco_R1]
#for ssh_login in all_login:
print(f"\nConnecting to {cisco_SW1['ip']}")
net_connect = ConnectHandler(**cisco_SW1)

for vlan in range(80, 90):
    print("Create VLAN " + str(vlan))

    config_commands = [
        "vlan " + str(vlan),
        "name MY_VLAN_" + str(vlan)
    ]
    output = net_connect.send_config_set(config_commands)
    print(output)

net_connect.disconnect()
print(f"\nConnecting to {cisco_R1['ip']}")
net_connect = ConnectHandler(**cisco_R1)
net_connect.disconnect()
